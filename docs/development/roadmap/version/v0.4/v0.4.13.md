```python
class OperationalMetrics:
    """Operational metrics collection for production."""

    def __init__(self):
        self.query_count = 0
        self.error_count = 0
        self.latencies = []
        self.cache_hits = 0
        self.cache_misses = 0

    def record_query(self, latency_ms: float, success: bool):
        """Record query metrics."""
        self.query_count += 1
        if success:
            self.latencies.append(latency_ms)
        else:
            self.error_count += 1
            logger.error(
                "Query failed",
                extra={
                    "metric": "query_error",
                    "latency_ms": latency_ms,
                    "error_rate": self.error_rate()
                }
            )

    def error_rate(self) -> float:
        """Calculate error rate."""
        if self.query_count == 0:
            return 0.0
        return self.error_count / self.query_count

    def get_health_metrics(self) -> dict:
        """Get health metrics for monitoring."""
        return {
            "queries_total": self.query_count,
            "errors_total": self.error_count,
            "error_rate": self.error_rate(),
            "latency_p50": np.percentile(self.latencies, 50) if self.latencies else 0,
            "latency_p95": np.percentile(self.latencies, 95) if self.latencies else 0,
            "latency_p99": np.percentile(self.latencies, 99) if self.latencies else 0,
            "cache_hit_rate": self.cache_hits / (self.cache_hits + self.cache_misses)
                if (self.cache_hits + self.cache_misses) > 0 else 0
        }
```

```python
class HealthChecker:
    """Comprehensive health check system."""

    def run_startup_checks(self) -> HealthCheckResult:
        """Run all startup health checks.

        Validates system readiness before accepting traffic.
        """
        checks = [
            self.check_configuration(),
            self.check_backend_connectivity(),
            self.check_database_health(),
            self.check_graph_health(),
            self.check_storage_capacity(),
            self.check_permissions(),
            self.check_dependencies()
        ]

        failed_checks = [c for c in checks if not c.passed]

        return HealthCheckResult(
            healthy=len(failed_checks) == 0,
            checks=checks,
            failed_checks=failed_checks
        )

    def check_backend_connectivity(self) -> HealthCheck:
        """Verify vector backend is accessible and functional."""
        try:
            # Test query
            backend = get_vector_backend()
            test_vector = np.random.rand(384)
            results = backend.query(test_vector, k=1)

            return HealthCheck(
                name="Backend Connectivity",
                passed=len(results) > 0,
                message="Backend responding to queries",
                latency_ms=...
            )
        except Exception as e:
            return HealthCheck(
                name="Backend Connectivity",
                passed=False,
                message=f"Backend error: {str(e)}",
                error=e
            )

    def check_database_health(self) -> HealthCheck:
        """Verify SQLite database is healthy."""
        try:
            # Check database file exists and is readable
            db_path = Path("~/.ragged/memory/ragged.db").expanduser()
            if not db_path.exists():
                return HealthCheck(
                    name="Database Health",
                    passed=False,
                    message="Database file not found"
                )

            # Check database integrity
            conn = sqlite3.connect(db_path)
            cursor = conn.cursor()
            cursor.execute("PRAGMA integrity_check")
            result = cursor.fetchone()[0]

            return HealthCheck(
                name="Database Health",
                passed=result == "ok",
                message=f"SQLite integrity: {result}"
            )
        except Exception as e:
            return HealthCheck(
                name="Database Health",
                passed=False,
                message=f"Database error: {str(e)}",
                error=e
            )

    def check_storage_capacity(self) -> HealthCheck:
        """Check available storage capacity."""
        ragged_dir = Path("~/.ragged").expanduser()
        stat = os.statvfs(ragged_dir)

        # Calculate available space
        available_gb = (stat.f_bavail * stat.f_frsize) / (1024**3)
        threshold_gb = 1.0  # Warn if <1GB available

        return HealthCheck(
            name="Storage Capacity",
            passed=available_gb > threshold_gb,
            message=f"Available: {available_gb:.1f} GB",
            warning=available_gb < threshold_gb * 2
        )

    def run_liveness_probe(self) -> bool:
        """Quick liveness check (is the process alive?)."""
        try:
            # Minimal check - can we respond?
            return True
        except:
            return False

    def run_readiness_probe(self) -> bool:
        """Readiness check (can we handle requests?)."""
        critical_checks = [
            self.check_backend_connectivity(),
            self.check_database_health()
        ]

        return all(check.passed for check in critical_checks)
```

```python
@contextmanager
def backend_unavailable():
    """Simulate backend unavailability."""
    original_backend = ragged._vector_backend
    try:
        ragged._vector_backend = UnavailableBackendMock()
        yield
    finally:
        ragged._vector_backend = original_backend

@contextmanager
def disk_full():
    """Simulate full disk."""
    with patch('os.statvfs') as mock_statvfs:
        mock_stat = MagicMock()
        mock_stat.f_bavail = 0  # No space available
        mock_statvfs.return_value = mock_stat
        yield

@contextmanager
def memory_pressure(available_mb: int):
    """Simulate memory pressure."""
    # Force cache evictions
    with patch.object(LRUCache, 'maxsize', available_mb // 10):
        yield
```
