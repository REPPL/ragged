```python
    # Try built-in parser
    result = _parse_builtin(expr)
    if result:

    try:
        import dateparser
        parsed = dateparser.parse(expr)
        if parsed:

    except ImportError:
        pass
```

```python
def _extract_vectors_chromadb(
    self,
    persona: Optional[str],
    batch_size: int
) -> Iterator[List[Dict[str, Any]]]:
    """Extract vectors from ChromaDB."""
    import chromadb

    client = chromadb.PersistentClient(path=str(self.chroma_path))

    # Get collections to migrate
    collections = self._get_collections(client, persona)

    for collection in collections:
        offset = 0
        while True:
            # Fetch batch
            results = collection.get(
                limit=batch_size,
                offset=offset,
                include=["embeddings", "metadatas", "documents"]
            )

            if not results["ids"]:
                break

            # Format for migration
            batch = []
            for i, doc_id in enumerate(results["ids"]):
                batch.append({
                    "id": doc_id,
                    "embedding": results["embeddings"][i],
                    "metadata": results["metadatas"][i],
                    "document": results["documents"][i],
                    "collection": collection.name,
                })

            yield batch
            offset += batch_size
```

```python
def _load_vectors_chromadb(
    self,
    vectors: List[Dict[str, Any]]
) -> None:
    """Load vectors into ChromaDB."""
    import chromadb

    client = chromadb.PersistentClient(path=str(self.chroma_path))

    # Group by collection
    by_collection = {}
    for vec in vectors:
        coll_name = vec["collection"]
        if coll_name not in by_collection:
            by_collection[coll_name] = []
        by_collection[coll_name].append(vec)

    # Load each collection
    for coll_name, vecs in by_collection.items():
        collection = client.get_or_create_collection(coll_name)

        collection.upsert(
            ids=[v["id"] for v in vecs],
            embeddings=[v["embedding"] for v in vecs],
            metadatas=[v["metadata"] for v in vecs],
            documents=[v["document"] for v in vecs],
        )
```

```python
def _create_backup(
    self,
    backend: str,
    persona: Optional[str]
) -> Path:
    """Create backup before migration."""
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    backup_name = f"backup_{backend}_{persona or 'all'}_{timestamp}"
    backup_path = self.backup_dir / backup_name

    if backend == "chromadb":
        # Copy ChromaDB directory
        shutil.copytree(self.chroma_path, backup_path / "chromadb")
    elif backend == "leann":
        # Copy LEANN index files
        shutil.copytree(self.leann_path, backup_path / "leann")

    # Compress backup
    archive_path = shutil.make_archive(
        str(backup_path), 'gztar', root_dir=backup_path
    )

    # Remove uncompressed backup
    shutil.rmtree(backup_path)

    logger.info(f"Backup created: {archive_path}")
    return Path(archive_path)
```

**Features:**
- Automatic compression (saves 80-90% space)
- Retention: Keep last 5 backups, delete older
- CLI: `ragged backend backup --backend chromadb`

```python
def rollback_migration(
    self,
    backup_path: Path,
    backend: str
) -> None:
    """Rollback migration using backup."""
    logger.warning(f"Rolling back {backend} from {backup_path}")

    # Extract backup
    extract_dir = backup_path.parent / "restore_temp"
    shutil.unpack_archive(str(backup_path), extract_dir)

    # Replace current backend data
    if backend == "chromadb":
        shutil.rmtree(self.chroma_path)
        shutil.move(extract_dir / "chromadb", self.chroma_path)
    elif backend == "leann":
        shutil.rmtree(self.leann_path)
        shutil.move(extract_dir / "leann", self.leann_path)

    # Cleanup
    shutil.rmtree(extract_dir)

    logger.info(f"Rollback complete: {backend}")
```

**Safety:**
- Verify backup integrity before rollback
- Atomic replacement (temp directory swap)
- Post-rollback verification

#### 6.3 Enhanced Migration Verification (3-4 hours)

**Purpose:** Verify migration beyond simple count matching

**Current:** `_verify_migration()` returns True (placeholder)

**Enhanced Verification:**
1. **Document count matching** (already planned)
2. **Vector similarity sampling** (NEW)
   - Sample 100 random documents
   - Compare vector embeddings (cosine similarity)
   - Ensure similarity > 0.99 (floating-point precision tolerance)
3. **Metadata integrity** (NEW)

4. **Query result matching** (NEW)

```python
def _verify_migration(
    self,
    source: str,
    target: str,
    persona: Optional[str]
) -> bool:
    """Comprehensive migration verification."""
    logger.info("Verifying migration...")

    # 1. Count matching
    source_count = self._get_document_count(source, persona)
    target_count = self._get_document_count(target, persona)

    if source_count != target_count:
        logger.error(f"Count mismatch: {source_count} vs {target_count}")
        return False

    # 2. Vector similarity sampling
    sample_ids = self._sample_documents(source, persona, n=100)
    for doc_id in sample_ids:
        source_vec = self._get_vector(source, doc_id)
        target_vec = self._get_vector(target, doc_id)

        similarity = cosine_similarity(source_vec, target_vec)
        if similarity < 0.99:
            logger.error(f"Vector mismatch for {doc_id}: {similarity}")
            return False

    # 3. Metadata integrity
    for doc_id in sample_ids:
        source_meta = self._get_metadata(source, doc_id)
        target_meta = self._get_metadata(target, doc_id)

        if source_meta != target_meta:
            logger.error(f"Metadata mismatch for {doc_id}")
            return False

    # 4. Query result matching
    test_queries = self._generate_test_queries(source, persona, n=10)
    for query in test_queries:
        source_results = self._query_backend(source, query, top_k=10)
        target_results = self._query_backend(target, query, top_k=10)

        if source_results != target_results:
            logger.warning(f"Query result mismatch: {query}")
            # Non-fatal (ranking may differ slightly due to precision)

    logger.info("Migration verification passed ✓")
    return True
```

```python
import click
from ragged.backend.migration import BackendMigrator

@click.group()
def backend():
    """Backend management commands."""
    pass

@backend.command()
@click.option("--from", "source", required=True, type=click.Choice(["chromadb", "leann"]))
@click.option("--to", "target", required=True, type=click.Choice(["chromadb", "leann"]))
@click.option("--persona", default=None, help="Migrate specific persona")
@click.option("--batch-size", default=100, help="Documents per batch")
@click.option("--no-verify", is_flag=True, help="Skip verification")
@click.option("--dry-run", is_flag=True, help="Estimate only")
@click.option("--resume", is_flag=True, help="Resume from checkpoint")
def migrate(source, target, persona, batch_size, no_verify, dry_run, resume):
    """Migrate vectors between backends."""
    migrator = BackendMigrator()

    if dry_run:
        # Estimate time and size
        estimate = migrator.estimate_migration(source, target, persona)
        click.echo(f"Estimated documents: {estimate['document_count']}")
        click.echo(f"Estimated time: {estimate['time_minutes']} minutes")
        click.echo(f"Storage change: {estimate['size_change_mb']} MB")
        return

    # Run migration with progress bar
    with click.progressbar(length=100, label="Migrating") as bar:
        result = migrator.migrate(
            source, target, persona, batch_size,
            verify=not no_verify,
            resume=resume,
            progress_callback=lambda pct: bar.update(pct)
        )

    click.echo(f"✓ Migrated {result.documents_migrated} documents in {result.duration_seconds:.1f}s")
```

```python
    latencies = []
    for query in queries:
        start = time.perf_counter()
        results = query_backend(backend, query, top_k=10)
        latency = time.perf_counter() - start
        latencies.append(latency)

    return {
        "p50": np.percentile(latencies, 50),
        "p95": np.percentile(latencies, 95),
        "p99": np.percentile(latencies, 99),
        "mean": np.mean(latencies),
    }
```
