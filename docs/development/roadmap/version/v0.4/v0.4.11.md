#### Time Expression Parsing with dateparser

**Natural Language Time Expressions**:
- "last week" → datetime range
- "-30d" → 30 days ago
- "January" → 2026-01-01 to 2026-01-31
- "Q4 2025" → 2025-10-01 to 2025-12-31
- "this morning" → today 00:00 to 12:00
- "yesterday at 3pm" → specific timestamp
- "in 2 weeks" → future date

**Relative Time**:
- "yesterday", "today", "tomorrow"
- "this week", "last month", "next year"
- "3 days ago", "2 weeks from now"
- "beginning of last quarter"
- "end of this month"

**Absolute Time**:
- "2026-01-15"
- "March 2025"
- "Q3 2024"
- "15th of January 2026"
- "1st quarter 2025"

#### Temporal Reasoning Implementation

```python
class TemporalReasoner:
    """Apply temporal reasoning to queries using dateparser."""

    def __init__(self):
        """Initialise temporal reasoner with dateparser."""
        # Configure dateparser for ragged's needs
        self.parser_settings = {
            'PREFER_DATES_FROM': 'past',  # Assume past unless specified
            'RETURN_AS_TIMEZONE_AWARE': True,
            'TIMEZONE': 'UTC',
            'RELATIVE_BASE': datetime.now(timezone.utc)
        }

    def parse_time_expression(
        self,
        expression: str,
        reference_time: Optional[datetime] = None
    ) -> TimeRange:
        """Parse natural language time expression.

        Uses dateparser for robust parsing of diverse time expressions.

        Examples:
        - "last week" → TimeRange(start=7d ago 00:00, end=today 00:00)
        - "January 2026" → TimeRange(start=2026-01-01, end=2026-01-31 23:59:59)
        - "Q4 2025" → TimeRange(start=2025-10-01, end=2025-12-31 23:59:59)
        - "yesterday at 3pm" → TimeRange(single point: yesterday 15:00)
        """
        import dateparser

        # Update relative base if provided
        settings = self.parser_settings.copy()
        if reference_time:
            settings['RELATIVE_BASE'] = reference_time

        # Parse expression
        parsed = dateparser.parse(expression, settings=settings)

        if parsed is None:
            raise TimeParseError(f"Could not parse time expression: {expression}")

        # Determine if point in time or range
        if self._is_range_expression(expression):
            return self._expand_to_range(parsed, expression)
        else:
            return TimeRange(start=parsed, end=parsed)

    def enhance_query_with_time(
        self,
        query: str,
        persona: str
    ) -> EnhancedQuery:
        """Add temporal context to query.

        Examples:
        - "my recent work" → Focus on last 30d
        - "when I was at Company X" → Filter by employment period
        - "my old notes on Python" → Emphasise older content
        - "what was I learning last month?" → Filter to last month
        """
        # Extract temporal expressions from query
        time_expressions = self._extract_time_expressions(query)

        # Parse all time expressions
        time_contexts = []
        for expr in time_expressions:
            try:
                time_range = self.parse_time_expression(expr)
                time_contexts.append(time_range)
            except TimeParseError:
                # Skip unparseable expressions
                continue

        # Get temporal facts relevant to query
        relevant_facts = self._get_relevant_temporal_facts(query, persona)

        # Build enhanced query
        return EnhancedQuery(
            original_query=query,
            time_contexts=time_contexts,
            temporal_facts=relevant_facts,
            focus_period=self._determine_focus_period(query, time_contexts)
        )

    def apply_temporal_filter(
        self,
        results: list[Document],
        time_context: TimeContext
    ) -> list[Document]:
        """Filter results by temporal context.

        Args:
            results: Documents to filter
            time_context: Temporal context with time ranges

        Returns:
            Filtered documents with temporal relevance
        """
        filtered = []
        for doc in results:
            # Check if document timestamp within time context
            if time_context.includes_time(doc.timestamp):
                # Calculate temporal relevance score
                relevance = self.calculate_temporal_relevance(
                    doc, time_context
                )
                doc.temporal_score = relevance
                filtered.append(doc)

        return sorted(filtered, key=lambda d: d.temporal_score, reverse=True)

    def calculate_recency_score(
        self,
        document: Document,
        query_time: datetime,
        decay_function: str = "exponential"
    ) -> float:
        """Calculate recency relevance score.

        Args:
            document: Document to score
            query_time: Time of query
            decay_function: "exponential", "linear", or "logarithmic"

        Returns:
            Recency score [0.0, 1.0]
        """
        age_days = (query_time - document.timestamp).total_seconds() / 86400

        if decay_function == "exponential":
            # Recent documents heavily favoured
            return math.exp(-age_days / 30)  # 30-day half-life
        elif decay_function == "linear":
            # Linear decay over 365 days
            return max(0.0, 1.0 - (age_days / 365))
        elif decay_function == "logarithmic":
            # Slower decay for older content
            return 1.0 / (1.0 + math.log1p(age_days / 30))
        else:
            raise ValueError(f"Unknown decay function: {decay_function}")

    def _extract_time_expressions(self, query: str) -> list[str]:
        """Extract potential time expressions from query.

        Uses pattern matching and dateparser's language understanding.
        """
        # Common time expression patterns
        patterns = [
            r'\b(last|this|next)\s+(week|month|year|quarter)\b',
            r'\b\d+\s+(days?|weeks?|months?|years?)\s+ago\b',
            r'\b(yesterday|today|tomorrow)\b',
            r'\b(jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)[a-z]*\s+\d{4}\b',
            r'\bQ[1-4]\s+\d{4}\b',
            r'\bin\s+\d{4}\b',
            # Add more patterns as needed
        ]

        expressions = []
        for pattern in patterns:
            matches = re.finditer(pattern, query, re.IGNORECASE)
            expressions.extend([m.group(0) for m in matches])

        return expressions

    def _is_range_expression(self, expression: str) -> bool:
        """Determine if expression represents a range vs point in time."""
        range_keywords = [
            'week', 'month', 'year', 'quarter',
            'january', 'february', 'march', 'april', 'may', 'june',
            'july', 'august', 'september', 'october', 'november', 'december',
            'q1', 'q2', 'q3', 'q4'
        ]
        return any(keyword in expression.lower() for keyword in range_keywords)

    def _expand_to_range(self, parsed_date: datetime, expression: str) -> TimeRange:
        """Expand single date to appropriate range based on expression."""
        expr_lower = expression.lower()

        if 'month' in expr_lower or any(month in expr_lower for month in [
            'january', 'february', 'march', 'april', 'may', 'june',
            'july', 'august', 'september', 'october', 'november', 'december'
        ]):
            # Month range
            start = parsed_date.replace(day=1, hour=0, minute=0, second=0)
            next_month = (start.month % 12) + 1
            year = start.year if next_month > start.month else start.year + 1
            end = start.replace(month=next_month, year=year) - timedelta(seconds=1)
            return TimeRange(start=start, end=end)

        elif 'week' in expr_lower:
            # Week range (Monday to Sunday)
            start = parsed_date - timedelta(days=parsed_date.weekday())
            start = start.replace(hour=0, minute=0, second=0)
            end = start + timedelta(days=7) - timedelta(seconds=1)
            return TimeRange(start=start, end=end)

        elif 'year' in expr_lower:
            # Year range
            start = parsed_date.replace(month=1, day=1, hour=0, minute=0, second=0)
            end = start.replace(year=start.year + 1) - timedelta(seconds=1)
            return TimeRange(start=start, end=end)

        elif any(q in expr_lower for q in ['q1', 'q2', 'q3', 'q4']):
            # Quarter range
            quarter = next(int(q[1]) for q in ['q1', 'q2', 'q3', 'q4'] if q in expr_lower)
            start_month = (quarter - 1) * 3 + 1
            start = parsed_date.replace(month=start_month, day=1, hour=0, minute=0, second=0)
            end_month = start_month + 2
            end = start.replace(month=end_month + 1) - timedelta(seconds=1)
            return TimeRange(start=start, end=end)

        else:
            # Default: single day
            start = parsed_date.replace(hour=0, minute=0, second=0)
            end = start + timedelta(days=1) - timedelta(seconds=1)
            return TimeRange(start=start, end=end)
```

class TimelineVisualiser:
    """Create ASCII visualisations for temporal data."""

    def render_timeline(
        self,
        timeline: Timeline,
        width: int = 60
    ) -> str:
        """Render activity timeline as ASCII chart."""
        max_activity = max(day.query_count for day in timeline.days)
        lines = []

        for day in timeline.days:
            # Calculate bar length
            bar_length = int((day.query_count / max_activity) * 10)
            bar = '█' * bar_length + '░' * (10 - bar_length)

            # Format day line
            date_str = day.date.strftime('%a %m/%d')
            count_str = f"{day.query_count} queries"
            topics_str = ", ".join(day.top_topics[:3]) if day.top_topics else "-"

            line = f"{date_str}  {bar}  {count_str:12}  Topics: {topics_str}"
            lines.append(line)

        return "\n".join(lines)

    def render_trend_graph(
        self,
        topic: str,
        evolution: TopicEvolution,
        width: int = 60
    ) -> str:
        """Render topic evolution as ASCII graph."""
        max_queries = max(week.query_count for week in evolution.weeks)
        lines = []

        for i, week in enumerate(evolution.weeks, start=1):
            # Calculate bar
            bar_length = int((week.query_count / max_queries) * 10)
            bar = '█' * bar_length + '░' * (10 - bar_length)

            # Check for peak or trend
            marker = ""
            if week.query_count == max_queries:
                marker = "← Peak"
            elif i == len(evolution.weeks) and week.is_trending_up:
                marker = "← Trending up"

            # Format line
            week_label = f"Week {i:2d}"
            count = f"{week.query_count:3d}"
            date = week.start_date.strftime("(%b %d)")

            line = f"{week_label}   {bar}  {count}  {date}  {marker}"
            lines.append(line)

        return "\n".join(lines)

    def render_comparison(
        self,
        comparison: PeriodComparison
    ) -> str:
        """Render period comparison with bars and change indicators."""
        lines = []

        for topic, stats in comparison.topic_changes.items():
            # Calculate bars
            max_count = max(stats.period_a_count, stats.period_b_count, 1)
            bar_a_len = int((stats.period_a_count / max_count) * 14)
            bar_b_len = int((stats.period_b_count / max_count) * 14)

            bar_a = '█' * bar_a_len + '░' * (14 - bar_a_len)
            bar_b = '█' * bar_b_len + '░' * (14 - bar_b_len)

            # Change indicator
            change_pct = stats.change_percentage
            if change_pct > 50:
                indicator = "↑↑"
            elif change_pct > 0:
                indicator = "↑"
            elif change_pct < -50:
                indicator = "↓↓"
            elif change_pct < 0:
                indicator = "↓"
            else:
                indicator = "→"

            # Format line
            change_str = f"{change_pct:+3.0f}%"
            line = f"  {topic:20s}  {bar_a}  {bar_b}    {change_str:6s} {indicator}"
            lines.append(line)

        return "\n".join(lines)

    def render_heatmap(
        self,
        heatmap_data: ActivityHeatmap,
        granularity: str = "week"
    ) -> str:
        """Render activity heatmap."""
        # Implementation for weekly heatmap
        lines = []
        header = "         Mon  Tue  Wed  Thu  Fri  Sat  Sun"
        lines.append(header)

        for week_num, week_data in enumerate(heatmap_data.weeks, start=1):
            day_cells = []
            for day in week_data:
                if day.query_count == 0 and not day.has_data:
                    cell = "  -  "
                elif day.query_count <= 2:
                    cell = " ░░░ "
                elif day.query_count <= 6:
                    cell = " ██░ "
                else:
                    cell = " ███ "
                day_cells.append(cell)

            week_label = f"Week {week_num:2d}"
            line = f"{week_label}  " + " ".join(day_cells)
            lines.append(line)

        return "\n".join(lines)

    2. Document count never negative

    valid_from: datetime,
    validity_days: int
):
    """Property: Temporal facts maintain validity invariants.

    Properties tested:
    1. valid_from always before valid_to
    2. Facts retrievable within validity period
    3. Facts not retrievable outside validity period
    4. Version numbers monotonically increasing
    """
    valid_to = valid_from + timedelta(days=validity_days)

    # Create fact
    fact = temporal_store.add_fact(
        persona="test",
        fact_type="test",
        content=fact_content,
        valid_from=valid_from,
        valid_to=valid_to
    )

    # Property 1: valid_from < valid_to
    assert fact.valid_from < fact.valid_to

    # Property 2: Retrievable within period
    midpoint = valid_from + timedelta(days=validity_days // 2)
    facts_at_midpoint = temporal_store.get_facts_at("test", midpoint)
    assert fact.id in [f.id for f in facts_at_midpoint]

    # Property 3: Not retrievable outside period
    before = valid_from - timedelta(days=1)
    after = valid_to + timedelta(days=1)
    facts_before = temporal_store.get_facts_at("test", before)
    facts_after = temporal_store.get_facts_at("test", after)
    assert fact.id not in [f.id for f in facts_before]
    assert fact.id not in [f.id for f in facts_after]


@given(
    expression=st.sampled_from([
        "last week", "this month", "yesterday", "Q1 2026",
        "January 2026", "3 days ago", "next month"
    ])
)
def test_time_parsing_properties(expression: str):
    """Property: Time expression parsing is consistent and valid.

    Properties tested:
    1. Same expression always parses to same result
    2. Parsed ranges have valid start/end
    3. Range duration makes sense for expression
    """
    # Parse twice
    range1 = reasoner.parse_time_expression(expression)
    range2 = reasoner.parse_time_expression(expression)

    # Property 1: Deterministic
    assert range1 == range2

    # Property 2: Valid range
    assert range1.start <= range1.end

    # Property 3: Reasonable duration
    duration_days = (range1.end - range1.start).days

    if "week" in expression:
        assert 6 <= duration_days <= 7
    elif "month" in expression or any(month in expression.lower() for month in [
        "january", "february", "march", "april", "may", "june",
        "july", "august", "september", "october", "november", "december"
    ]):
        assert 27 <= duration_days <= 31
    elif "quarter" in expression.lower() or expression.startswith("Q"):
        assert 88 <= duration_days <= 92


@given(
    num_facts=st.integers(min_value=0, max_value=100),
    persona=st.text(min_size=1, max_size=20, alphabet=st.characters(whitelist_categories=('Ll', 'Lu')))
)
def test_fact_versioning_properties(num_facts: int, persona: str):
    """Property: Fact versioning maintains consistency.

    Properties tested:
    1. Version numbers start at 1
    2. Version numbers increment by 1
    3. Version history complete (no gaps)
    4. Latest version matches current fact
    """
    # Create and update facts
    facts = []
    for i in range(num_facts):
        fact = temporal_store.add_fact(
            persona=persona,
            fact_type="test",
            content=f"Fact {i}",
            valid_from=datetime.now()
        )
        facts.append(fact)

        # Update fact random number of times
        num_updates = random.randint(0, 5)
        for v in range(num_updates):
            temporal_store.update_fact(
                fact.id,
                content=f"Fact {i} version {v+2}"
            )

    # Check properties for each fact
    for fact in facts:
        history = temporal_store.get_fact_history(fact.id)

        # Property 1: Starts at 1
        assert history[0].version == 1

        # Property 2: Increments by 1
        versions = [v.version for v in history]
        assert versions == list(range(1, len(versions) + 1))

        # Property 3: No gaps
        assert len(set(versions)) == len(versions)

        # Property 4: Latest matches current
        current = temporal_store.get_fact(fact.id)
        assert history[-1].content == current.content
