## Overview

Comprehensive backend infrastructure release combining migration tools with performance optimisation. This release ensures users can seamlessly migrate between vector backends (ChromaDB ↔ LEANN) while achieving production-ready performance at scale.

**Vision**: Users can confidently choose and switch between backends based on their needs, with robust migration tools and comprehensive performance benchmarking. The memory system performs efficiently at scale (10K+ documents) with minimal latency overhead.

**Rationale**: With LEANN (v0.4.3) offering 97% storage savings but platform-specific availability, users need both migration tools AND performance optimisation to make informed decisions and achieve production-grade performance regardless of backend choice.

**Two-Part Integration**: This release merges previously separate backend migration (old v0.4.11) and performance optimisation (old v0.4.12) into a cohesive infrastructure release that addresses both backend selection and performance tuning.

##### Migration Architecture

```python
class BackendMigrator:
    """Migrate vector data between backends with safety features."""

    def migrate(
        self,
        source_backend: str,
        target_backend: str,
        persona: Optional[str] = None,
        batch_size: int = 100,
        verify: bool = True,
        dry_run: bool = False,  # NEW: Dry-run mode
        checkpoint_interval: int = 1000  # NEW: Checkpointing
    ) -> MigrationResult:
        """Migrate from source to target backend.

        Args:
            source_backend: "chromadb" or "leann"
            target_backend: "chromadb" or "leann"
            persona: Migrate specific persona (None = all)
            batch_size: Documents per batch
            verify: Verify migration integrity
            dry_run: Estimate time/storage without actually migrating
            checkpoint_interval: Save checkpoint every N documents

        Returns:
            Migration result with statistics
        """
        # Dry-run mode: estimate without migrating
        if dry_run:
            return self._dry_run_estimate(
                source_backend, target_backend, persona
            )

        # 1. Validate backends available
        self._validate_backends(source_backend, target_backend)

        # 2. Create backup
        backup_path = self._create_backup(source_backend, persona)

        # 3. Extract vectors from source (with checkpointing)
        checkpoint_file = None
        vectors_migrated = 0

        try:
            for batch_num, vectors in enumerate(
                self._extract_vectors(source_backend, persona, batch_size)
            ):
                # 4. Load vectors into target
                self._load_vectors(target_backend, vectors, batch_size)
                vectors_migrated += len(vectors)

                # 5. Checkpoint progress
                if (batch_num + 1) % (checkpoint_interval // batch_size) == 0:
                    checkpoint_file = self._save_checkpoint(
                        source_backend, target_backend, persona, vectors_migrated
                    )

            # 6. Verify migration (if enabled)
            if verify:
                verification = self._verify_migration(
                    source_backend, target_backend, persona
                )
                if not verification.success:
                    self._rollback(backup_path, target_backend)
                    raise MigrationError(verification.errors)

            # 7. Run automated regression detection
            regression_check = self._detect_regressions(
                source_backend, target_backend, persona
            )
            if not regression_check.passed:
                logger.warning(
                    f"Performance regressions detected: {regression_check.issues}"
                )

            # 8. Update configuration
            self._update_backend_config(target_backend, persona)

            # 9. Cleanup checkpoint file
            if checkpoint_file:
                os.remove(checkpoint_file)

            return MigrationResult(
                source=source_backend,
                target=target_backend,
                documents_migrated=vectors_migrated,
                personas_migrated=[persona] if persona else self._all_personas(),
                duration_seconds=...,
                verification=verification,
                regression_check=regression_check,
                backup_path=backup_path
            )

        except Exception as e:
            # Automatic rollback on error
            if checkpoint_file:
                logger.info(f"Migration failed. Checkpoint saved: {checkpoint_file}")
                logger.info("Resume with: ragged backend migrate --resume {checkpoint_file}")
            raise

    def _dry_run_estimate(
        self,
        source: str,
        target: str,
        persona: Optional[str]
    ) -> DryRunEstimate:
        """Estimate migration time and storage without migrating.

        Returns:
            Estimated duration, storage changes, compatibility warnings
        """
        # Count documents
        doc_count = self._count_documents(source, persona)

        # Estimate duration based on benchmarks
        avg_docs_per_sec = 50  # Conservative estimate
        estimated_seconds = doc_count / avg_docs_per_sec

        # Estimate storage
        current_storage = self._measure_storage(source, persona)
        target_storage_estimate = self._estimate_storage(target, doc_count)
        storage_change = target_storage_estimate - current_storage

        # Check compatibility
        warnings = self._check_compatibility(source, target)

        return DryRunEstimate(
            document_count=doc_count,
            estimated_duration_seconds=estimated_seconds,
            current_storage_mb=current_storage,
            target_storage_mb=target_storage_estimate,
            storage_change_mb=storage_change,
            storage_change_pct=(storage_change / current_storage) * 100,
            warnings=warnings
        )

    def _detect_regressions(
        self,
        source: str,
        target: str,
        persona: Optional[str]
    ) -> RegressionCheck:
        """Detect performance regressions after migration.

        Runs benchmark queries on both backends and compares:
        - Query latency (should be within 20%)
        - Recall accuracy (should be >95% similar)
        - Memory usage
        """
        # Run sample queries on both backends
        test_queries = self._generate_test_queries(persona, count=20)

        source_results = self._benchmark_queries(source, test_queries)
        target_results = self._benchmark_queries(target, test_queries)

        issues = []

        # Check latency regression
        latency_change = (
            (target_results.mean_latency - source_results.mean_latency)
            / source_results.mean_latency
        )
        if latency_change > 0.20:  # >20% slower
            issues.append(
                f"Query latency increased by {latency_change*100:.1f}%"
            )

        # Check recall consistency
        recall_similarity = self._compare_recall(
            source_results.documents, target_results.documents
        )
        if recall_similarity < 0.95:
            issues.append(
                f"Recall similarity only {recall_similarity*100:.1f}%"
            )

        return RegressionCheck(
            passed=len(issues) == 0,
            issues=issues,
            source_latency_ms=source_results.mean_latency,
            target_latency_ms=target_results.mean_latency,
            recall_similarity=recall_similarity
        )

    def resume_migration(
        self,
        checkpoint_file: str
    ) -> MigrationResult:
        """Resume interrupted migration from checkpoint."""
        checkpoint = self._load_checkpoint(checkpoint_file)

        # Resume from where we left off
        return self.migrate(
            source_backend=checkpoint.source,
            target_backend=checkpoint.target,
            persona=checkpoint.persona,
            batch_size=checkpoint.batch_size,
            verify=checkpoint.verify,
            resume_from_document=checkpoint.last_document_id
        )
```

class BackendSelector:
    """Select optimal backend for platform and constraints."""

    def recommend_backend(
        self,
        storage_constraint: Optional[int] = None,  # MB
        platform_override: Optional[str] = None
    ) -> BackendRecommendation:
        """Recommend backend for user's platform.

        Considers:
        - Platform compatibility (macOS, Linux, Windows)
        - Storage constraints
        - Current document count
        - Performance requirements
        """
        platform = platform_override or self._detect_platform()

        # Check LEANN availability
        leann_available = self._check_leann_available(platform)

        # Estimate storage needs
        current_docs = self._count_documents()
        estimated_storage = {
            "chromadb": current_docs * 200,  # ~200KB per doc
            "leann": current_docs * 20       # ~20KB per doc (90% savings)
        }

        # Make recommendation with reasoning
        if not leann_available:
            return BackendRecommendation(
                backend="chromadb",
                reason="LEANN not available on platform",
                platform=platform,
                storage_estimate=estimated_storage["chromadb"]
            )

        if storage_constraint and estimated_storage["chromadb"] > storage_constraint:
            return BackendRecommendation(
                backend="leann",
                reason=f"Storage savings needed ({estimated_storage['leann']}MB vs {estimated_storage['chromadb']}MB)",
                platform=platform,
                storage_estimate=estimated_storage["leann"],
                storage_savings=estimated_storage["chromadb"] - estimated_storage["leann"]
            )

        # Default: ChromaDB for simplicity unless storage constrained
        return BackendRecommendation(
            backend="chromadb",
            reason="Default backend (adequate storage available)",
            platform=platform,
            storage_estimate=estimated_storage["chromadb"],
            alternative="leann",
            alternative_benefit=f"Switch to LEANN to save {estimated_storage['chromadb'] - estimated_storage['leann']}MB"
        )

##### Graph Traversal Optimisation

```python
class OptimisedLEANNBackend:
    """LEANN backend with query optimisations and plan caching."""

    def __init__(self, config: LEANNConfig):
        self.graph = self._initialise_graph(config)
        self.cache = LRUCache(maxsize=config.cache_size)
        self.query_planner = QueryPlanner()
        self.plan_cache = LRUCache(maxsize=1000)  # NEW: Plan caching

    def query(
        self,
        query_vector: np.ndarray,
        k: int,
        filters: Optional[dict] = None
    ) -> list[Document]:
        """Optimised query with caching and plan caching."""

        # 1. Check result cache
        cache_key = self._cache_key(query_vector, k, filters)
        if cache_key in self.cache:
            return self.cache[cache_key]

        # 2. Query planning with plan caching
        plan_key = self._plan_cache_key(k, filters, len(self.graph.nodes))
        if plan_key in self.plan_cache:
            plan = self.plan_cache[plan_key]
        else:
            plan = self.query_planner.plan(query_vector, k, filters, self.graph)
            self.plan_cache[plan_key] = plan

        # 3. Execute with optimisations
        if plan.use_approximate:
            results = self._approximate_search(query_vector, k, plan)
        else:
            results = self._exact_search(query_vector, k, plan)

        # 4. Cache results
        self.cache[cache_key] = results

        return results

    def _approximate_search(
        self,
        query_vector: np.ndarray,
        k: int,
        plan: QueryPlan
    ) -> list[Document]:
        """Approximate graph search with early termination.

        Optimisations:
        - Beam search instead of exhaustive traversal
        - Early termination when k good candidates found
        - Distance-based pruning
        - Adaptive beam width
        """
        beam_width = min(k * 3, 100)  # Adaptive beam width
        candidates = []

        # Start from entry points (optimised selection)
        entry_points = self._select_entry_points(query_vector, plan)

        for entry in entry_points:
            # Beam search from entry point
            visited = set()
            current_beam = [(entry, self._distance(query_vector, entry))]

            while current_beam and len(candidates) < k * 2:
                # Expand beam
                next_beam = []
                for node, dist in current_beam[:beam_width]:
                    if node in visited:
                        continue
                    visited.add(node)

                    # Add to candidates
                    candidates.append((node, dist))

                    # Explore neighbours
                    for neighbour in self.graph.neighbours(node):
                        if neighbour not in visited:
                            neighbour_dist = self._distance(query_vector, neighbour)
                            next_beam.append((neighbour, neighbour_dist))

                # Sort and prune beam
                next_beam.sort(key=lambda x: x[1])
                current_beam = next_beam[:beam_width]

                # Early termination
                if self._early_termination_condition(candidates, k):
                    break

        # Return top-k
        candidates.sort(key=lambda x: x[1])
        return [node for node, _ in candidates[:k]]

    def batch_query(
        self,
        query_vectors: list[np.ndarray],
        k: int
    ) -> list[list[Document]]:
        """Optimised batch querying.

        Optimisations:
        - Shared graph traversals
        - Vectorised distance calculations
        - Parallel processing (threading)
        """
        # Group similar queries for shared traversal
        query_groups = self._group_similar_queries(query_vectors)

        results = []
        with ThreadPoolExecutor(max_workers=4) as executor:
            futures = []
            for group in query_groups:
                future = executor.submit(
                    self._batch_query_group, group, k
                )
                futures.append(future)

            for future in as_completed(futures):
                results.extend(future.result())

        return results
```

class MemorySystemCache:
    """Multi-level caching for memory system."""

    def __init__(self, config: CacheConfig):
        # L1: In-memory cache (hot data)
        self.l1_cache = LRUCache(maxsize=config.l1_size)

        # L2: Profile cache (user profiles)
        self.profile_cache = TTLCache(
            maxsize=config.l2_size,
            ttl=config.profile_ttl
        )

        # L3: Query result cache
        self.query_cache = TTLCache(
            maxsize=config.l3_size,
            ttl=config.query_ttl
        )

    def get_profile(self, persona: str) -> Optional[InterestProfile]:
        """Get profile with multi-level cache."""
        # Check L1
        if persona in self.l1_cache:
            return self.l1_cache[persona]

        # Check L2
        if persona in self.profile_cache:
            profile = self.profile_cache[persona]
            self.l1_cache[persona] = profile  # Promote to L1
            return profile

        return None  # Cache miss
```

class MemoryBenchmark:
    """Comprehensive memory system benchmarking with load testing."""

    def run_full_benchmark(
        self,
        backends: list[str],
        corpus_sizes: list[int],
        queries_per_test: int = 100,
        include_load_test: bool = False  # NEW: Load testing
    ) -> BenchmarkReport:
        """Run complete benchmark suite.

        Tests:
        1. Ingest performance
        2. Query latency (single, batch)
        3. Memory usage
        4. Storage efficiency
        5. Concurrent access
        6. Personalised ranking overhead
        7. Temporal query performance
        8. Load testing (optional)
        """

        results = {}
        for backend in backends:
            results[backend] = {}
            for size in corpus_sizes:
                results[backend][size] = {
                    "ingest": self._benchmark_ingest(backend, size),
                    "query": self._benchmark_query(backend, queries_per_test),
                    "memory": self._benchmark_memory(backend, size),
                    "storage": self._measure_storage(backend, size),
                    "concurrent": self._benchmark_concurrent(backend),
                    "personalisation": self._benchmark_personalisation(backend),
                    "temporal": self._benchmark_temporal(backend)
                }

                # Optional load testing
                if include_load_test:
                    results[backend][size]["load_test"] = self._run_load_test(
                        backend, size, duration_minutes=5
                    )

        return BenchmarkReport(results)

    def _run_load_test(
        self,
        backend: str,
        corpus_size: int,
        duration_minutes: int = 5
    ) -> LoadTestResults:
        """Run load test with sustained query load.

        Simulates production load:
        - Multiple concurrent users
        - Sustained query rate
        - Mix of query types
        - Measure degradation over time
        """
        num_users = 10
        queries_per_sec = 5
        duration_sec = duration_minutes * 60

        results = {
            "throughput": [],
            "latency_p95": [],
            "memory_usage": []
        }

        start_time = time.time()
        with ThreadPoolExecutor(max_workers=num_users) as executor:
            while time.time() - start_time < duration_sec:
                # Submit queries
                futures = []
                for _ in range(queries_per_sec):
                    query = self._generate_random_query()
                    future = executor.submit(
                        self._execute_query, backend, query
                    )
                    futures.append(future)

                # Collect results
                latencies = [f.result() for f in futures]
                results["latency_p95"].append(np.percentile(latencies, 95))
                results["throughput"].append(len(futures))
                results["memory_usage"].append(self._current_memory_usage())

                time.sleep(1)

        return LoadTestResults(
            duration_seconds=duration_sec,
            total_queries=sum(results["throughput"]),
            avg_throughput=np.mean(results["throughput"]),
            avg_latency_p95=np.mean(results["latency_p95"]),
            max_memory_mb=max(results["memory_usage"]),
            performance_degradation=self._calculate_degradation(results)
        )
```

#### 7. Profiling & Monitoring Tools (2-3h)

Runtime profiling and monitoring for production deployments.

```python
class PerformanceProfiler:
    """Profile ragged performance in production."""

    def enable(self, components: list[str] = None):
        """Enable profiling for components.

        Args:
            components: ["query", "memory", "backend", "all"]
        """
        self.enabled = True
        self.components = components or ["all"]
        self.metrics = defaultdict(list)

    def profile_query(self, query: str, persona: str):
        """Profile query execution."""
        with self.timer("query_total"):
            # Profile each stage
            with self.timer("query.enhance"):
                enhanced = enhance_with_context(query, persona)

            with self.timer("query.retrieve"):
                results = retrieve(enhanced, k=20)

            with self.timer("query.rerank"):
                reranked = personalised_rerank(results, persona, k=10)

            with self.timer("query.generate"):
                response = generate(query, reranked)

        return response

    def get_report(self) -> PerformanceReport:
        """Get performance report."""
        return PerformanceReport({
            component: {
                "count": len(timings),
                "mean": np.mean(timings),
                "p50": np.percentile(timings, 50),
                "p95": np.percentile(timings, 95),
                "p99": np.percentile(timings, 99),
            }
            for component, timings in self.metrics.items()
        })
```

**CLI Profiling**:
```bash

# Run queries with profiling

# Get profiling report
ragged profile report --since "-1h"
```
